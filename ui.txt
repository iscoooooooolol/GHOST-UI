local function tw(obj, props, time, style, dir)
    return TweenService:Create(obj,
        TweenInfo.new(time or 0.18, style or Enum.EasingStyle.Quart, dir or Enum.EasingDirection.Out),
        props):Play()
end

local function corner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = parent
    return c
end

local function stroke(parent, color, thick, transp)
    local s = Instance.new("UIStroke")
    s.Color = color or C.BORDER
    s.Thickness = thick or 1
    s.Transparency = transp or 0.4
    s.Parent = parent
    return s
end

local function newFrame(props)
    local f = Instance.new("Frame")
    f.BackgroundColor3 = props.bg or C.CARD
    f.BorderSizePixel  = 0
    f.Size             = props.size or UDim2.new(1,0,0,40)
    f.Position         = props.pos or UDim2.new(0,0,0,0)
    f.LayoutOrder      = props.order or 0
    if props.parent then f.Parent = props.parent end
    if props.transparent then f.BackgroundTransparency = 1 end
    return f
end

local function newLabel(props)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Font      = props.font or Enum.Font.GothamMedium
    l.Text      = props.text or ""
    l.TextColor3 = props.color or C.TEXT
    l.TextSize  = props.size or 12
    l.TextXAlignment = props.align or Enum.TextXAlignment.Left
    l.TextYAlignment = props.valign or Enum.TextYAlignment.Center
    l.Size      = props.sz or UDim2.new(1,0,1,0)
    l.Position  = props.pos or UDim2.new(0,0,0,0)
    l.TextWrapped = props.wrap or false
    if props.parent then l.Parent = props.parent end
    return l
end

    end)

        if inp.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end

-- Tab method
function Ghost:Tab(name, icon)
    local tabObj = {}
    tabObj._name = name
    tabObj._cards = {}
    tabObj._window = self

    -- Nav button
    local navBtn = Instance.new("TextButton")
    navBtn.Size = UDim2.new(1,0,0,38)
    navBtn.BackgroundColor3 = C.SIDEBAR
    navBtn.BorderSizePixel = 0
    navBtn.Text = ""
    navBtn.AutoButtonColor = false
    navBtn.LayoutOrder = #self._tabs + 1
    navBtn.Parent = self._navList
    corner(navBtn, 8)

    -- Active indicator bar
    local activeBar = newFrame({
        bg=C.ACCENT,
        size=UDim2.new(0,3,0,20),
        pos=UDim2.new(0,0,0.5,-10),
        parent=navBtn
    })
    corner(activeBar, 3)
    activeBar.Visible = false

    -- Icon
    local iconL = newLabel({
        text  = ICONS[icon] or icon or "•",
        sz    = 15,
        color = C.TEXT_DIM,
        align = Enum.TextXAlignment.Center,
        pos   = UDim2.new(0,10,0,0),
        parent= navBtn,
    })
    iconL.Size = UDim2.new(0,26,1,0)
    iconL.Position = UDim2.new(0,8,0,0)

    -- Label
    local nameL = newLabel({
        text  = name,
        sz    = 12,
        color = C.TEXT_DIM,
        font  = Enum.Font.GothamMedium,
        pos   = UDim2.new(0,40,0,0),
        parent= navBtn,
    })
    nameL.Size = UDim2.new(1,-44,1,0)
    nameL.Position = UDim2.new(0,40,0,0)

    -- Group all cards in a folder tag
    tabObj._folder = name

    local function activate()
        if self._activeTab == tabObj then return end

        if self._activeBtn then
            local prevBtn = self._activeBtn
            tw(prevBtn, {BackgroundColor3=C.SIDEBAR}, 0.15)
            prevBtn:FindFirstChild("ActiveBar") and (prevBtn.ActiveBar.Visible = false)
            local prevIcon = prevBtn:FindFirstChild("Icon")
            local prevName = prevBtn:FindFirstChild("Name")
            if prevIcon then tw(prevIcon, {TextColor3=C.TEXT_DIM}, 0.15) end
            if prevName then tw(prevName, {TextColor3=C.TEXT_DIM}, 0.15) end
        end

        self._activeTab = tabObj
        self._activeBtn = navBtn
        activeBar.Visible = true
        tw(navBtn, {BackgroundColor3=Color3.fromRGB(32,35,38)}, 0.15)
        tw(iconL, {TextColor3=C.ACCENT}, 0.15)
        tw(nameL, {TextColor3=C.TEXT}, 0.15)

        -- Show/hide cards
        for _, card in ipairs(self._scroll:GetChildren()) do
            if card:IsA("Frame") then
                card.Visible = (card:GetAttribute("Tab") == name)
            end
        end
    end

    tabObj._activate = activate

    navBtn.Name = "NavBtn_"..name
    activeBar.Name = "ActiveBar"
    iconL.Name = "Icon"
    nameL.Name = "Name"

    navBtn.MouseEnter:Connect(function()
        if self._activeTab ~= tabObj then
            tw(navBtn, {BackgroundColor3=Color3.fromRGB(28,30,34)}, 0.12)
        end
    end)
    navBtn.MouseLeave:Connect(function()
        if self._activeTab ~= tabObj then
            tw(navBtn, {BackgroundColor3=C.SIDEBAR}, 0.12)
        end
    end)
    navBtn.MouseButton1Click:Connect(activate)

    table.insert(self._tabs, tabObj)

    -- Activate first tab automatically
    if #self._tabs == 1 then
        task.defer(activate)
    end

    function tabObj:Card(title, cardIcon)
        local win = self._window
        win._cardOrder = win._cardOrder + 1

        })
        header.BackgroundTransparency = 1

        local hPad = Instance.new("UIPadding")
        hPad.PaddingLeft = UDim.new(0,14)
        hPad.PaddingRight = UDim.new(0,14)

            })
            corner(ddBg, 7)
            stroke(ddBg, C.BORDER, 1, 0.5)

            local selLabel = newLabel({
                text  = selected,
                sz    = 11,
                color = C.TEXT_MID,
                font  = Enum.Font.Gotham,
                pos   = UDim2.new(0,10,0,0),
                parent= ddBg,
            })
            selLabel.Size = UDim2.new(1,-30,1,0)
            selLabel.Position = UDim2.new(0,10,0,0)

            local arrow = newLabel({
                text  = "∧",
                sz    = 11,
                color = C.ACCENT,
                align = Enum.TextXAlignment.Center,
                parent= ddBg,
            })
            arrow.Size = UDim2.new(0,24,1,0)
            arrow.Position = UDim2.new(1,-26,0,0)
            arrow.Rotation = 180

            local open = false
            local optionFrames = {}

            local optionContainer = newFrame({
                bg=Color3.fromRGB(24,26,30),
                size=UDim2.new(1,0,0,0),
                pos=UDim2.new(0,0,0,52),
                parent=row
            })
            corner(optionContainer, 7)
            stroke(optionContainer, C.BORDER, 1, 0.5)
            optionContainer.ClipsDescendants = true
            optionContainer.Visible = false

            local optList = Instance.new("UIListLayout")
            optList.SortOrder = Enum.SortOrder.LayoutOrder
            optList.Parent = optionContainer

            for i, opt in ipairs(options) do
                local optBtn = Instance.new("TextButton")
                optBtn.Size = UDim2.new(1,0,0,28)
                optBtn.BackgroundTransparency = 1
                optBtn.BorderSizePixel = 0
                optBtn.Text = ""
                optBtn.LayoutOrder = i
                optBtn.Parent = optionContainer

                local optLbl = newLabel({
                    text  = opt,
                    sz    = 11,
                    color = C.TEXT_MID,
                    font  = Enum.Font.Gotham,
                    pos   = UDim2.new(0,10,0,0),
                    parent= optBtn,
                })
                optLbl.Size = UDim2.new(1,-10,1,0)
                optLbl.Position = UDim2.new(0,10,0,0)

                optBtn.MouseEnter:Connect(function()
                    tw(optBtn, {BackgroundTransparency=0.7}, 0.1)
                    optBtn.BackgroundColor3 = C.ACCENT_DIM
                end)
                optBtn.MouseLeave:Connect(function()
                    tw(optBtn, {BackgroundTransparency=1}, 0.1)
                end)
                optBtn.MouseButton1Click:Connect(function()
                    selected = opt
                    selLabel.Text = opt
                    open = false
                    optionContainer.Visible = false
                    arrow.Rotation = 180
                    if callback then task.spawn(callback, opt) end
                end)
                table.insert(optionFrames, optBtn)
            end

            local function toggleDD()
                open = not open
                optionContainer.Visible = open
                optionContainer.Size = UDim2.new(1,0,0, open and (optList.AbsoluteContentSize.Y) or 0)
                arrow.Rotation = open and 0 or 180
            end

            local ddBtn = Instance.new("TextButton")
            ddBtn.Size = UDim2.new(1,0,0,30)
            ddBtn.Position = UDim2.new(0,0,0,20)
            ddBtn.BackgroundTransparency = 1
            ddBtn.Text = ""
            ddBtn.Parent = row
            ddBtn.MouseButton1Click:Connect(toggleDD)
        end

        -- Text input row
        function cardObj:Input(label, placeholder, callback)
            rowOrder = rowOrder + 1
            local row = newFrame({
                bg=C.CARD, size=UDim2.new(1,0,0,24),
                parent=card, order=rowOrder, transparent=true
            })
            row.AutomaticSize = Enum.AutomaticSize.Y

            local rPad = Instance.new("UIPadding")
            rPad.PaddingLeft = UDim.new(0,14)
            rPad.PaddingRight = UDim.new(0,14)
            rPad.PaddingTop = UDim.new(0,4)
            rPad.Parent = row

            local lbl = newLabel({
                text  = label,
                sz    = 10,
                color = C.TEXT_DIM,
                font  = Enum.Font.Gotham,
                parent= row,
            })
            lbl.Size = UDim2.new(1,0,0,18)

            local inputBg = newFrame({
                bg=Color3.fromRGB(22,24,27),
                size=UDim2.new(1,0,0,30),
                pos=UDim2.new(0,0,0,20),
                parent=row
            })
            corner(inputBg, 7)
            stroke(inputBg, C.BORDER, 1, 0.5)

            local tb = Instance.new("TextBox")
            tb.Size = UDim2.new(1,-20,1,0)
            tb.Position = UDim2.new(0,10,0,0)
            tb.BackgroundTransparency = 1
            tb.Font = Enum.Font.Gotham
            tb.PlaceholderText = placeholder or ""
            tb.PlaceholderColor3 = C.TEXT_DIM
            tb.Text = ""
            tb.TextColor3 = C.TEXT
            tb.TextSize = 11
            tb.TextXAlignment = Enum.TextXAlignment.Left
            tb.ClearTextOnFocus = false
            tb.Parent = inputBg

            tb.FocusLost:Connect(function()
                if callback then task.spawn(callback, tb.Text) end
            end)

            inputBg.InputBegan:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    stroke(inputBg, C.ACCENT, 1, 0)
                end
            end)
            inputBg.InputEnded:Connect(function(inp)
                if inp.UserInputType == Enum.UserInputType.MouseButton1 then
                    stroke(inputBg, C.BORDER, 1, 0.5)
                end
            end)
        end

        -- Button
        function cardObj:Button(label, callback)
            rowOrder = rowOrder + 1
            local row = newFrame({
                bg=C.CARD, size=UDim2.new(1,0,0,36),
                parent=card, order=rowOrder, transparent=true
            })

            local rPad = Instance.new("UIPadding")
            rPad.PaddingLeft = UDim.new(0,14)
            rPad.PaddingRight = UDim.new(0,14)
            rPad.PaddingTop = UDim.new(0,4)
            rPad.Parent = row

            local btn = Instance.new("TextButton")
            btn.Size = UDim2.new(1,0,0,28)
            btn.BackgroundColor3 = Color3.fromRGB(32,36,40)
            btn.BorderSizePixel = 0
            btn.Text = label
            btn.TextColor3 = C.TEXT
            btn.TextSize = 11
            btn.Font = Enum.Font.GothamMedium
            btn.AutoButtonColor = false
            btn.Parent = row
            corner(btn, 7)
            stroke(btn, C.BORDER, 1, 0.4)

            btn.MouseEnter:Connect(function()
                tw(btn, {BackgroundColor3=Color3.fromRGB(40,46,52)}, 0.12)
                tw(btn, {TextColor3=C.ACCENT}, 0.12)
            end)
            btn.MouseLeave:Connect(function()
                tw(btn, {BackgroundColor3=Color3.fromRGB(32,36,40)}, 0.12)
                tw(btn, {TextColor3=C.TEXT}, 0.12)
            end)
            btn.MouseButton1Click:Connect(function()
                tw(btn, {BackgroundColor3=C.ACCENT_DIM}, 0.08)
                task.delay(0.12, function()
                    tw(btn, {BackgroundColor3=Color3.fromRGB(32,36,40)}, 0.12)
                    tw(btn, {TextColor3=C.TEXT}, 0.08)
                end)
                if callback then task.spawn(callback) end
            end)
        end

        -- Section divider inside card
        function cardObj:Divider(label)
            rowOrder = rowOrder + 1
            local row = newFrame({
                bg=C.CARD, size=UDim2.new(1,0,0,22),
                parent=card, order=rowOrder, transparent=true
            })

            local rPad = Instance.new("UIPadding")
            rPad.PaddingLeft = UDim.new(0,14)
            rPad.PaddingRight = UDim.new(0,14)
            rPad.Parent = row

            local div = newFrame({
                bg=C.BORDER, size=UDim2.new(1,0,0,1),
                pos=UDim2.new(0,0,0.5,0), parent=row
            })
            div.BackgroundTransparency = 0.55

            if label then
                local lDiv = newFrame({
                    bg=C.CARD,
                    size=UDim2.new(0, #label*6+12, 1,0),
                    pos=UDim2.new(0,0,0,0),
                    parent=row
                })
                newLabel({
                    text=label, sz=9, color=C.TEXT_DIM,
                    font=Enum.Font.GothamMedium,
                    align=Enum.TextXAlignment.Left,
                    parent=lDiv,
                })
            end
        end

        return cardObj
    end

    return tabObj
end

-- Notification
function Ghost:Notify(title, content, duration)
    duration = duration or 3
    local gui = self._gui

    local notif = newFrame({
        bg=C.CARD, size=UDim2.new(0,280,0,64),
        pos=UDim2.new(1,-296,1,80), parent=gui
    })
    corner(notif, 10)
    stroke(notif, C.ACCENT, 1, 0.3)

    local accentBar = newFrame({
        bg=C.ACCENT, size=UDim2.new(0,3,1,-16),
        pos=UDim2.new(0,0,0,8), parent=notif
    })
    corner(accentBar, 3)

    newLabel({
        text=title, sz=12, font=Enum.Font.GothamBold,
        color=C.TEXT, pos=UDim2.new(0,16,0,10), parent=notif,
    }).Size = UDim2.new(1,-20,0,18)

    newLabel({
        text=content, sz=11, color=C.TEXT_DIM, font=Enum.Font.Gotham,
        pos=UDim2.new(0,16,0,30), parent=notif, wrap=true,
    }).Size = UDim2.new(1,-20,0,26)

    tw(notif, {Position=UDim2.new(1,-296,1,-80)}, 0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out)

    task.delay(duration, function()
        tw(notif, {Position=UDim2.new(1,-296,1,80)}, 0.25)
        task.delay(0.3, function() notif:Destroy() end)
    end)
end

return Ghost
